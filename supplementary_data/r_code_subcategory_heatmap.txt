library(ggplot2)
library(readr)
library(dplyr)

# Read data
df <- read_csv(
  "heatmap_top15_subcategories_long.csv",
  show_col_types = FALSE
)

# Subcategory abbreviations
abbr <- c(
  "Frameworks & Methodologies"      = "FM",
  "Internet Forensics"              = "IF",
  "FS Datalayer Analysis"           = "FS-DA",
  "FS Metadata Analysis"            = "FS-MA",
  "Memory Analysis"                 = "MemA",
  "Memory Acquisition"              = "MemAcq",
  "Mobile Forensics"                = "MobF",
  "IoT Devices"                     = "IoT",
  "Software-Generated Artifacts"    = "SGA",
  "Vehicle / Automotive Forensics"  = "VAF",
  "AI-Assisted Digital Forensics"   = "AIDF",
  "Cloud Forensics"                 = "CF",
  "Education and Training Papers"   = "ETP",
  "Reviews, Surveys and Ontologies" = "RSO",
  "OS-Level Memory Analysis"        = "OSMA"
)

# Clean and abbreviate
df <- df %>%
  mutate(
    subcategory = trimws(subcategory),
    subcategory = recode(subcategory, !!!abbr),
    published_year = as.integer(published_year),
    count = replace_na(count, 0)
  )

# Order subcategories by total paper count
subcategory_order <- df %>%
  group_by(subcategory) %>%
  summarise(total_count = sum(count, na.rm = TRUE), .groups = "drop") %>%
  arrange(desc(total_count)) %>%
  pull(subcategory)

df <- df %>%
  mutate(
    subcategory = factor(
      subcategory,
      levels = rev(subcategory_order)
    )
  )

# Create heatmap
p <- ggplot(
  df,
  aes(
    x = factor(published_year),
    y = subcategory,
    fill = count
  )
) +
  geom_tile(
    color = "white",
    linewidth = 0.2
  ) +
  scale_fill_gradientn(
    colours = c(
      "#fff7ec",
      "#fdd49e",
      "#fdbb84",
      "#e34a33",
      "#c51b7d",
      "#7f0000"
    ),
    name = "Papers"
  ) +
  labs(
    x = "Publication Year",
    y = "Subcategory"
  ) +
  theme_minimal(
    base_family = "serif",
    base_size = 14
  ) +
  theme(
    text = element_text(family = "serif"),
    panel.grid = element_blank(),

    axis.text.x = element_text(
      angle = 90,
      hjust = 1,
      vjust = 0.5,
      size = 11
    ),

    axis.text.y = element_text(
      size = 12
    ),

    axis.title.x = element_text(
      face = "plain",
      size = 14
    ),

    axis.title.y = element_text(
      face = "plain",
      size = 14
    ),

    legend.title = element_text(
      face = "plain",
      size = 13
    ),

    legend.text = element_text(
      size = 12
    ),

    plot.margin = margin(10, 10, 10, 10)
  )

# Display plot
print(p)

# Save PNG
ggsave(
  filename = "top15_subcategories_heatmap.png",
  plot = p,
  width = 12,
  height = 7,
  dpi = 600,
  bg = "white"
)

# Save PDF
ggsave(
  filename = "top15_subcategories_heatmap.pdf",
  plot = p,
  width = 12,
  height = 7,
  bg = "white"
)
